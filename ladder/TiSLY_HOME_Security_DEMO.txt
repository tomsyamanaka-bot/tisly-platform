;==============================================================================
; プロジェクト名 : TiSLY_HOME_Security_DEMO
; 用途           : TiSLY HOME Security デモ用 PLC ラダープログラム
; PLC            : 三菱電機 FX 系 (FX3U / FX5U 想定)
; 開発ツール     : GX Works2 / GX Works3
; 作成日         : 2026-05-28
; バージョン     : 1.0.0
;
; 【I/O 割り当て】
; 入力 X0 : セレクタスイッチ / 警戒 ON
; 入力 X1 : 非常停止ボタン (ON = 停止要求)
; 入力 X2 : ビームセンサー1 / 外周検知
; 入力 X3 : ビームセンサー2 / 近接検知
; 出力 Y0 : 24V 赤ライト
; 出力 Y1 : 100V 白ライト 1
; 出力 Y2 : 100V 白ライト 2
; 出力 Y3 : 100V 白ライト 3
; 出力 Y4 : 100V 白ライト 4
;
; 【内部リレー】
; M0  : 警戒中
; M1  : センサー1 警報保持
; M2  : センサー2 警報保持
; M10 : 低速点滅用 (将来拡張 / テンプレート予約)
; M11 : 高速点滅用 (将来拡張 / テンプレート予約)
; M20 : Y0 赤ライト制御 (二重コイル回避用)
;
; 【特殊補助リレー】
; M8013 : 1 秒クロック (0.5s ON / 0.5s OFF)
; M8012 : 0.1 秒クロック (0.05s ON / 0.05s OFF)
;==============================================================================

;------------------------------------------------------------------------------
; 段1 : 警戒 ON 保持
; コメント : セレクタ ON かつ非常停止でない場合、警戒状態 M0 を SET
;------------------------------------------------------------------------------
0     LD     X0          ; セレクタスイッチ / 警戒 ON
      ANI    X1          ; 非常停止でない (X1=OFF)
      SET    M0          ; 警戒中

;------------------------------------------------------------------------------
; 段2 : 警戒 OFF リセット
; コメント : セレクタ OFF で警戒・警報保持をすべて解除
;------------------------------------------------------------------------------
      LDI    X0          ; セレクタ OFF
      RST    M0          ; 警戒中 解除
      RST    M1          ; センサー1 警報 解除
      RST    M2          ; センサー2 警報 解除

;------------------------------------------------------------------------------
; 段3 : 非常停止
; コメント : 非常停止 ON で内部状態と全出力を強制 OFF
;------------------------------------------------------------------------------
      LD     X1          ; 非常停止ボタン ON
      ZRST   M0   M2     ; M0～M2 一括リセット
      ZRST   Y0   Y4     ; Y0～Y4 一括 OFF

;------------------------------------------------------------------------------
; 段4 : センサー1 警報保持
; コメント : 警戒中に外周ビーム検知 → M1 を SET (保持)
;------------------------------------------------------------------------------
      LD     M0          ; 警戒中
      AND    X2          ; ビームセンサー1 / 外周検知
      SET    M1          ; センサー1 警報保持

;------------------------------------------------------------------------------
; 段5 : センサー2 警報保持
; コメント : 警戒中に近接ビーム検知 → M2 を SET (保持)
;------------------------------------------------------------------------------
      LD     M0          ; 警戒中
      AND    X3          ; ビームセンサー2 / 近接検知
      SET    M2          ; センサー2 警報保持

;------------------------------------------------------------------------------
; 段6 : Y0 赤ライト制御 (内部リレー M20)
; コメント : M2=ON 時は 0.1s 高速点滅を最優先
;            M2=OFF かつ M0=ON 時は 1s 低速点滅
;            単一 RUNG で M20 を駆動 (Y0 二重コイル回避)
;------------------------------------------------------------------------------
      LD     M2          ; 近接警報中
      AND    M8012       ; 0.1 秒クロック (高速)
      LD     M0          ; 警戒中
      ANI    M2          ; 近接警報でない
      AND    M8013       ; 1 秒クロック (低速)
      ORB                ; 高速 OR 低速
      OUT    M20         ; 赤ライト制御

;------------------------------------------------------------------------------
; 段6-2 : Y0 出力
; コメント : M20 経由で Y0 を単一コイル出力
;------------------------------------------------------------------------------
      LD     M20
      OUT    Y0          ; 24V 赤ライト

;------------------------------------------------------------------------------
; 段7 : Y1 白ライト1
; コメント : 外周警報 (M1) で常時点灯
;------------------------------------------------------------------------------
      LD     M1          ; センサー1 警報保持
      OUT    Y1          ; 100V 白ライト 1

;------------------------------------------------------------------------------
; 段8 : Y2 白ライト2
; コメント : 外周警報 (M1) で 1 秒クロック点滅
;------------------------------------------------------------------------------
      LD     M1          ; センサー1 警報保持
      AND    M8013       ; 1 秒クロック
      OUT    Y2          ; 100V 白ライト 2

;------------------------------------------------------------------------------
; 段9 : Y3 白ライト3
; コメント : 近接警報 (M2) で常時点灯
;------------------------------------------------------------------------------
      LD     M2          ; センサー2 警報保持
      OUT    Y3          ; 100V 白ライト 3

;------------------------------------------------------------------------------
; 段10 : Y4 白ライト4
; コメント : 近接警報 (M2) で常時点灯
;------------------------------------------------------------------------------
      LD     M2          ; センサー2 警報保持
      OUT    Y4          ; 100V 白ライト 4

;------------------------------------------------------------------------------
; プログラム終了
;------------------------------------------------------------------------------
      END
