;==============================================================================
; プロジェクト名 : TiSLY_HOME_Security_DEMO
; 用途           : TiSLY HOME Security DEMO（ホームセキュリティデモ）
; PLC            : 三菱 FX系（FX3U / FX5U 等） / GX Works2 / GX Works3
; プログラム名   : MAIN
; 作成日         : 2026-05-28
; 備考           : 入力はa接点（ON=信号あり）想定。非常停止X1はONで停止。
;                  Y0は内部リレーM20で1コイル化（二重コイル回避）。
;                  M10/M11は将来TiSLY標準テンプレ用に予約（本プログラム未使用）。
;==============================================================================

;------------------------------------------------------------------------------
; 段1：警戒ON保持
; X0（セレクタ/警戒ON）がON、かつ非常停止X1がOFFのとき M0 をSET
;------------------------------------------------------------------------------
0    LD     X0          ; セレクタスイッチ / 警戒ON
     ANI    X1          ; 非常停止でない（X1=OFF）
     SET    M0          ; 警戒中フラグ ON

;------------------------------------------------------------------------------
; 段2：警戒OFFリセット
; X0がOFFのとき、警戒・警報保持をすべて解除
;------------------------------------------------------------------------------
     LDI    X0          ; 警戒OFF
     RST    M0          ; 警戒中 解除
     RST    M1          ; センサー1警報 解除
     RST    M2          ; センサー2警報 解除

;------------------------------------------------------------------------------
; 段3：非常停止
; X1がONのとき、全状態リセット＋全出力強制OFF
;------------------------------------------------------------------------------
     LD     X1          ; 非常停止ボタン ON
     RST    M0
     RST    M1
     RST    M2
     RST    Y0          ; 24V 赤ライト OFF
     RST    Y1          ; 100V 白ライト1 OFF
     RST    Y2          ; 100V 白ライト2 OFF
     RST    Y3          ; 100V 白ライト3 OFF
     RST    Y4          ; 100V 白ライト4 OFF

;------------------------------------------------------------------------------
; 段4：センサー1警報保持（外周検知）
; 警戒中 M0=ON かつ ビームセンサー1 X2=ON で M1 をSET（保持）
;------------------------------------------------------------------------------
     LD     M0          ; 警戒中
     AND    X2          ; ビームセンサー1 / 外周検知
     SET    M1          ; センサー1警報保持

;------------------------------------------------------------------------------
; 段5：センサー2警報保持（近接検知）
; 警戒中 M0=ON かつ ビームセンサー2 X3=ON で M2 をSET（保持）
;------------------------------------------------------------------------------
     LD     M0          ; 警戒中
     AND    X3          ; ビームセンサー2 / 近接検知
     SET    M2          ; センサー2警報保持

;------------------------------------------------------------------------------
; 段6：Y0 赤ライト制御（M20経由・単一コイル）
; 優先順位: M2=ON → M8012(0.1s)高速点滅 / M2=OFFかつM0=ON → M8013(1s)低速点滅
; M2 ON 時は低速点滅より高速点滅を優先（ORBで1段に集約）
;------------------------------------------------------------------------------
     LD     M2          ; 近接警報中
     AND    M8012       ; 0.1秒クロック（高速点滅）
     LD     M0          ; 警戒中
     ANI    M2          ; 近接警報なし（M2 優先のため遮断）
     AND    M8013       ; 1秒クロック（低速点滅）
     ORB                ; 高速 OR 低速
     OUT    M20         ; Y0制御用内部リレー

;------------------------------------------------------------------------------
; 段6b：Y0 出力（M20のみから駆動・二重コイルなし）
;------------------------------------------------------------------------------
     LD     M20
     OUT    Y0          ; 24V 赤ライト

;------------------------------------------------------------------------------
; 段7：Y1 白ライト1
; センサー1警報 M1=ON で常時点灯
;------------------------------------------------------------------------------
     LD     M1
     OUT    Y1          ; 100V 白ライト1

;------------------------------------------------------------------------------
; 段8：Y2 白ライト2
; センサー1警報 M1=ON で1秒クロック点滅
;------------------------------------------------------------------------------
     LD     M1
     AND    M8013       ; 1秒クロック
     OUT    Y2          ; 100V 白ライト2

;------------------------------------------------------------------------------
; 段9：Y3 白ライト3
; センサー2警報 M2=ON で常時点灯
;------------------------------------------------------------------------------
     LD     M2
     OUT    Y3          ; 100V 白ライト3

;------------------------------------------------------------------------------
; 段10：Y4 白ライト4
; センサー2警報 M2=ON で常時点灯
;------------------------------------------------------------------------------
     LD     M2
     OUT    Y4          ; 100V 白ライト4

;------------------------------------------------------------------------------
; プログラム終了
;------------------------------------------------------------------------------
     END
